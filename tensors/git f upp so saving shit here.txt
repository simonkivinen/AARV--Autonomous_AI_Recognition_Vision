import torch
import torch.nn as nn
import torch.optim as optim
import torchvision
import torchvision.transforms as transforms

# 1. LADDA NER DATASETET (60 000 bilder på siffror 0-9)
transformering = transforms.Compose([transforms.ToTensor(), transforms.Normalize((0.5,), (0.5,))])
train_set = torchvision.datasets.MNIST(root='./data', train=True, download=True, transform=transformering)
train_loader = torch.utils.data.DataLoader(train_set, batch_size=64, shuffle=True)

# 2. DEFINIERA AI-NÄTVERKET
class MNISTNetwork(nn.Module):
    def __init__(self):
        super().__init__()
        self.flatten = nn.Flatten()
        self.layer1 = nn.Linear(784, 128) # 28x28 pixlar = 784 ingångar
        self.relu = nn.ReLU()
        self.layer2 = nn.Linear(128, 10)  # 10 utgångar (siffrorna 0-9)
        
    def forward(self, x):
        x = self.flatten(x)
        x = self.layer1(x)
        x = self.relu(x)
        return self.layer2(x)

model = MNISTNetwork()
criterion = nn.CrossEntropyLoss()
optimizer = optim.SGD(model.parameters(), lr=0.01)

# 3. TRÄNA MODELLEN (Kör igenom bilderna)
print("Träning påbörjad på 60 000 bilder...")
model.train()
for batch_idx, (images, labels) in enumerate(train_loader):
    optimizer.zero_grad()
    outputs = model(images) # 4D Tensor skickas in här!
    loss = criterion(outputs, labels)
    loss.backward()
    optimizer.step()
    
    if batch_idx % 200 == 0:
        print(f"Bilder bearbetade: {batch_idx * len(images)}/60000 - Loss: {loss.item():.4f}")

print("\n🎉 Träning klar! Din AI kan nu känna igen siffror.")
